\section*{Problema 3}
(2.0) Seleccione DOS ejercicios de los siguientes:
\begin{enumerate}[a)]
  \item Demuestre la identidad 
  \(\sum_{k=0}^n \binom{n}{k}^3 = \sum_{k=0}^n \binom{n}{k}^2 \binom{2k}{n}.\)
  \item Definimos \(T_n = \tfrac{n(n+1)}{2}\) y \(C(n,k)=\binom{n-2k}{k}\). Demuestre que
  \[\sum_{k=0}^{n-1} C(T_n,k)\,C(T_{n+1},k)
    = \frac{n^3 - 2n^2 + 4n}{n+2}\,
      \binom{T_n}{n}\,\binom{T_{n+1}}{n}.\]
  \item Para \(m\le n\), demuestre
  \[\sum_{k=m}^n (-1)^{k+m}\,\frac{2k+1}{\binom{n+k}{n-k}}\,
    \binom{2k}{k-m}
    = \frac{1}{2n+1}.\]
  \item Para \(n\ge0\), demuestre
  \[\sum_{k=0}^n (-1)^k \binom{2k}{k} \binom{2n-2k}{n-k}
    = \begin{cases}
      2^n \binom{n}{n/2}, & n\text{ par},\\
      0, & n\text{ impar}.
    \end{cases}\]
  \item Demuestre
  \[2\sum_{k=0}^\infty (-1)^k \binom{n+k}{k} \frac{2n+2k+1}{\binom{2n+2k}{2k}}
    = \binom{2n}{n}\sum_{k=0}^\infty \frac{2^k}{2n+2k+1}\binom{2n+2k}{n+k}.\]
  \item Para \(n>0\), encuentre fórmula cerrada:
  \[\sum_{k=0}^n \binom{n}{k}(2k-1)!!\,(2(n-k)-1)!!,\]
  donde \(m!! = m(m-2)\cdots\).
  \item Para \(m>0\), demuestre
  \[\sum_{k=0}^n \binom{n}{k} (-1)^k \frac{m+k}{m(m+1)\cdots(m+n)}
    = \frac{n!}{m(m+1)\cdots(m+n)}.\]
\end{enumerate}
