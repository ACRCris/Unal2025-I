\begin{solution}

    Sea
  \[
  x^4 + a x^3 + b x^2 + c x + d = 0
  \]
  y hagamos la sustitución
  \[
  x = y - \alpha,
  \qquad
  \alpha := \frac{a}{4}.
  \]

  Expandiendo cada uno de los términos, usando el triangulo de pascal.
  \begin{align*}
  x^4 &= (y - \alpha)^4
       = y^4 
         - 4\alpha\,y^3
         + 6\alpha^2\,y^2
         - 4\alpha^3\,y
         + \alpha^4, \\[6pt]
  a\,x^3 &= a\,(y-\alpha)^3
         = a\Bigl(y^3 - 3\alpha\,y^2 + 3\alpha^2\,y - \alpha^3\Bigr),\\[4pt]
  b\,x^2 &= b\,(y-\alpha)^2
         = b\bigl(y^2 - 2\alpha\,y + \alpha^2\bigr),\\[4pt]
  c\,x   &= c\,(y-\alpha)
         = c\,y - c\,\alpha,\\[4pt]
  d     &= d.
  \end{align*}


  Sumando los términos y simplificando:
  \begin{align*}
    x^4 + a x^3 + b x^2 + c x + d
    &= y^4
    \cancelto{0}{
      +\bigl[-4\alpha + a\bigr]\,y^3}
    + \bigl[6\alpha^2 - 3a\alpha + b\bigr]\,y^2
    \\&+ \bigl[-4\alpha^3 + 3a\alpha^2 - 2b\alpha + c\bigr]\,y
    + \bigl[\alpha^4 - a\alpha^3 + b\alpha^2 - c\alpha + d\bigr].  
  \end{align*}
  Pues tenemos que:
\begin{align*}
  -4\alpha + a  &= \;-4\Bigl(\tfrac{a}{4}\Bigr) + a \\\;&=\; -a + a \;=\;0.
\end{align*}
Por lo que ademas si definimos:
  \begin{align*}
  p &= 6\alpha^2 - 3a\alpha + b\\
    &= 6\bigl(\tfrac{a}{4}\bigr)^2 - 3a\bigl(\tfrac{a}{4}\bigr) + b\\
    &= b - \tfrac{3a^2}{8}, \\\\[4pt]
  q &= -4\alpha^3 + 3a\alpha^2 - 2b\alpha + c\\
    &= -4\bigl(\tfrac{a}{4}\bigr)^3 + 3a\bigl(\tfrac{a}{4}\bigr)^2 - 2b\bigl(\tfrac{a}{4}\bigr) + c\\
    &= \frac{a^3}{16}\left(
      -1 + 3
    \right) - \frac{2ab}{4} + c\\
    &= c - \tfrac{ab}{2} + \tfrac{a^3}{8}, \\\\[4pt]
  r &= \alpha^4 - a\alpha^3 + b\alpha^2 - c\alpha + d\\
    &= \bigl(\tfrac{a}{4}\bigr)^4 - a\bigl(\tfrac{a}{4}\bigr)^3 + b\bigl(\tfrac{a}{4}\bigr)^2 - c\bigl(\tfrac{a}{4}\bigr) + d\\
    &= \frac{a^4}{64}\left(
      \frac 14-1
    \right) + \frac{ba^2}{16} - \frac{ca}{4} + d\\
    &= d - \tfrac{ac}{4} + \tfrac{ba^2}{16} - \tfrac{3a^4}{256}.
  \end{align*}
  
  
  Por lo tanto, la ecuación se transforma en:  

  \[
  y^4 + \Bigl(b - \tfrac{3a^2}{8}\Bigr)\,y^2
       + \Bigl(c - \tfrac{ab}{2} + \tfrac{a^3}{8}\Bigr)\,y
       + \Bigl(d - \tfrac{ac}{4} + \tfrac{ba^2}{16} - \tfrac{3a^4}{256}\Bigr)
  = 0,
  \]
  es decir,
  \[
  y^4 + p\,y^2 + q\,y + r = 0,
  \]
    con \(p,q,r\) definidos arriba.  

   
  \end{solution}
