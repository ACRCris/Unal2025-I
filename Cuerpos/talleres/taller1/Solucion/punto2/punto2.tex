\begin{solution}
    \begin{align*}
        \textbf{Datos:}\quad 
            &g(t):=t^{6}-1, 
              &f(t):=t^{4}-1
              &&\bigl(g,f\in K[t],\;K\text{ cuerpo cualquiera}\bigr)
        \end{align*}
        
        \medskip
        \noindent\textbf{Algoritmo de Euclides (divisiones sucesivas).}
        
        \[
        \begin{aligned}
        \boxed{1}\;&
        g(t)=t^{2}\,f(t)+\underbrace{\bigl(t^{2}-1\bigr)}_{=:r_{1}(t)}
              &&\deg r_{1}<\deg f
              \\[4pt]
        \boxed{2}\;&
        f(t)=\bigl(t^{2}+1\bigr)\,r_{1}(t)+\underbrace{0}_{r_{2}(t)}
        \end{aligned}
        \]
        
        \noindent
        La \textit{última} resta no nula es $r_{1}(t)=t^{2}-1$, por lo que
        
        \[
        \gcd\!\bigl(g(t),f(t)\bigr)=t^{2}-1.
        \]
        
        \bigskip
        \noindent\textbf{Retro–sustitución (identidad de Bézout).}
        
        De la primera división:
        \[
        r_{1}(t)=t^{2}-1=g(t)-t^{2}\,f(t).
        \]
        
        Por lo tanto existen
        \[
        a(t)=-t^{2}, \qquad b(t)=1 \quad\bigl(a(t),b(t)\in K[t]\bigr)
        \]
        tales que
        \[
        \boxed{\;
        \gcd\!\bigl(g(t),f(t)\bigr)=t^{2}-1
            =a(t)\,f(t)+b(t)\,g(t)
            =\bigl(-t^{2}\bigr)\,f(t)+1\cdot g(t)
        \;} .
        \]
        
        \bigskip
        \noindent\textbf{Cadena de igualdades compacta.}
        \[
        \bigl(g(t),f(t)\bigr)
          =\bigl(f(t),g(t)-t^{2}f(t)\bigr)
          =\bigl(f(t),t^{2}-1\bigr)
          =\bigl(t^{2}-1,0\bigr)
          =t^{2}-1.
        \]
        
        \medskip\hrule\medskip
        
        \emph{Observación.}  
        En general, para cualquier par de enteros positivos $m,n$ y cualquier
        cuerpo $K$ se verifica  
        \(
        \gcd\!\bigl(t^{m}-1,\;t^{n}-1\bigr)=t^{\gcd(m,n)}-1,
        \)
        y el algoritmo precedente muestra el caso particular $m=6,\;n=4$.
\end{solution}
