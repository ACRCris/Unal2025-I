\begin{solucion}
    
  \item Para las raíces $\alpha^2,\beta^2,\gamma^2$, sea
  \[
  Q(y)=y^3 + A\,y^2 + B\,y + C.
  \]
  Usamos que
  \[
  \alpha^2+\beta^2+\gamma^2
  = (\alpha+\beta+\gamma)^2 - 2(\alpha\beta+\alpha\gamma+\beta\gamma)
  = (-2)^2 - 2(-3) = 10,
  \]
  \[
  \alpha^2\beta^2 + \alpha^2\gamma^2 + \beta^2\gamma^2
  = (\alpha\beta+\alpha\gamma+\beta\gamma)^2 - 2\,\alpha\beta\gamma\,(\alpha+\beta+\gamma)
  = (-3)^2 - 2(-5)(-2) = -11,
  \]
  \[
  \alpha^2\beta^2\gamma^2 = (\alpha\beta\gamma)^2 = (-5)^2 = 25.
  \]
  Luego,
  \[
  \begin{cases}
  \alpha^2 + \beta^2 + \gamma^2 = -A \;\Longrightarrow\;A=-10,\\
  \alpha^2\beta^2 + \alpha^2\gamma^2 + \beta^2\gamma^2 = B \;\Longrightarrow\;B=-11,\\
  \alpha^2\beta^2\gamma^2 = -C \;\Longrightarrow\;C=-25.
  \end{cases}
  \]
  Por lo tanto,
  \[
  Q(y)=y^3 - 10y^2 - 11y - 25.
  \]
\end{solucion}
